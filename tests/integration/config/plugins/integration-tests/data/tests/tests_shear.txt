test "Shear - Research Projects"
	status active
	description "Tests building a research lab, choosing a project, and finishing research, which reveals the shear."
	sequence
		call "Fabrication - Setup"
		# The rows are: the warehouse, the Orbital Station, the Alloy Kiln, the
		# Defense Platform, the Fabrication Bay, then the Research Lab.
		input
			key "Down"
		input
			key e
		assert
			"facility: Research Lab" == 1
		# Research view: the cheapest unfinished project, Deep Survey Array, comes first.
		input
			key "Tab"
		input
			key "Tab"
		input
			key e
		input
			key "F12"
		input
			key 9
		input
			key "escape"
		assert
			"research: Deep Survey Array" == 1
		# Now the list is Convoy Countermeasures, then Shear Mapping.
		input
			key "Down"
		input
			key e
		input
			key "F12"
		input
			key 9
		input
			key "escape"
		assert
			and
				"research: Shear Mapping" == 1
				"research: Convoy Countermeasures" == 0


test "Shear - Carrier Fold Jump"
	status active
	description "Tests that a saved carrier is placed on loading, and that ordering it across the shear while aboard takes the player along."
	sequence
		inject "Carrier At Suhail Save"
		call "Load First Savegame"
		assert
			and
				"flagship planet: Test Ark" > 0
				"carrier in: Suhail" == 1
		# Industry, then the carrier view. The fold jump is listed first.
		input
			key n
		input
			key "Tab"
		input
			key "Tab"
		input
			key "Tab"
		input
			key e
		label "launching"
		branch "launching"
			"flagship landed" > 0
		assert
			and
				"flagship system: Threshold" > 0
				"carrier in: Threshold" == 1
		navigate
			"travel destination" "Test Ark"
		call "Land"
		assert
			"flagship planet: Test Ark" > 0
